// Rotated DavisLUT geometry experiment -- interactive GUI runner.
//
// 1. Load validate_geom_gui.json through Configuration -> Load.
// 2. Open Scripting -> JavaScript and load this TXT file.
// 3. Edit the controls below and run.
//
// The script rebuilds the ideal interface and both monitors with a nontrivial
// rigid rotation, rotates the source beam consistently, and compares monitor-
// local theta/phi distributions with the original LUT coordinates.

var VALIDATION_DIR = "";
var DIRECTION = "fwd";     // "fwd" = LYSO -> air; "rev" = air -> LYSO
var ANGLE = 45.0;          // incidence angle from the rotated interface normal
var ROTATION = [37.0, 29.0, 23.0]; // ANTS3/TGeo [Phi, Theta, Psi], degrees
var PHOTONS = 300000;
var SEED = 20260726;

var FWD = "../luts/pooled_28um_fwd.lut";
var REV = "../luts/pooled_28um_rev.lut";
var OUT = "geom_gui_rotated_out";
var EXPECTED_CONFIG_NAME = "DavisLUT-GUI-validation";

if (typeof geo == "undefined" || typeof geowin == "undefined" ||
    typeof graph == "undefined" || typeof hist == "undefined" ||
    typeof grwin == "undefined")
    core.abort("This script requires GUI mode with geo/geowin/graph/hist interfaces");
if (VALIDATION_DIR != "") core.setCurrentDir(VALIDATION_DIR);
if (config.getKeyValue("ConfigName") != EXPECTED_CONFIG_NAME)
    core.abort("Load validate_geom_gui.json through Configuration -> Load first");
if (DIRECTION != "fwd" && DIRECTION != "rev")
    core.abort("DIRECTION should be 'fwd' or 'rev'");
if (ANGLE <= 0 || ANGLE >= 90)
    core.abort("ANGLE should be strictly between 0 and 90 degrees");
if (ROTATION.length != 3)
    core.abort("ROTATION should be [Phi, Theta, Psi]");
if (PHOTONS <= 0)
    core.abort("PHOTONS should be positive");
if (!core.isFileExist(FWD) || !core.isFileExist(REV))
    core.abort("Cannot find LUT files; start ANTS3 in validation/ or set VALIDATION_DIR");

var reverse = DIRECTION == "rev";
var lut = core.loadObject(reverse ? REV : FWD);
var EX = rotateVector([1, 0, 0], ROTATION);
var EY = rotateVector([0, 1, 0], ROTATION);
var NORMAL = rotateVector([0, 0, 1], ROTATION);
var normalSign = reverse ? -1.0 : 1.0;
var rad = ANGLE*Math.PI/180.0;
var depth = 0.002;
var source = add(scale(EX, -depth*Math.tan(rad)),
                 scale(NORMAL, -normalSign*depth));
var direction = add(scale(EX, Math.sin(rad)),
                    scale(NORMAL, normalSign*Math.cos(rad)));

core.createDir(OUT);
core.createDir(OUT + "/sim");
config.replace("PhotonSim.PhotonBombs.PhotonsPerBomb.FixedNumber", PHOTONS);
config.replace("PhotonSim.PhotonBombs.Single.Position", source);
config.replace("PhotonSim.PhGenOverrides.Direction.DirectionVector", direction);
config.replace("PhotonSim.Run.OutputDirectory", OUT + "/sim");
config.replace("PhotonSim.Run.SaveMonitors", true);
config.updateConfig();

// Rebuild the same ideal two-medium experiment in a rotated frame. Full box
// sizes are used by geo.box(). AirBox=40 mm keeps every tilted LYSO corner
// strictly inside its mother volume.
geo.clearWorld();
geo.box("AirBox", [40, 40, 40], 0, "World", [0, 0, 0], [0, 0, 0]);
geo.box("LYSO", [20, 20, 10], 1, "AirBox",
        scale(NORMAL, -5.0), ROTATION);
geo.monitor("LowerMonitor_reflection_or_transmission", 0, 12, 12, "LYSO",
            [0, 0, 4.996], [0, 0, 0], true, true, true);
geo.monitor("UpperMonitor_reflection_or_transmission", 0, 12, 12, "AirBox",
            scale(NORMAL, 0.004), ROTATION, true, true, true);
geo.configurePhotonMonitor("LowerMonitor_reflection_or_transmission",
                           [80, 80], [], [45, 0, 90], []);
geo.configurePhotonMonitor("UpperMonitor_reflection_or_transmission",
                           [80, 80], [], [45, 0, 90], []);
geo.updateGeometry(true);

if (lsim.countMonitors() != 2)
    core.abort("The rotated geometry should contain exactly two photon monitors");

// The material rules remain loaded from validate_geom_gui.json.
geowin.clearTracks();
geowin.redraw();
var displayLength = 4.0;
var displaySource = scale(direction, -displayLength);
geowin.addTrack([displaySource, [0, 0, 0]], 2, 1, 3); // red incident guide
geowin.showTracksAndMarkers();
geowin.showWindow(false);

lsim.setSeed(SEED);
lsim.simulate();
lsim.loadMonitorData(OUT + "/sim/PhotonMonitors.txt");

var reflectedIndex = reverse ? 1 : 0;
var transmittedIndex = reverse ? 0 : 1;
var measuredR = normalizeMonitor(lsim.getMonitorAngle(reflectedIndex));
var measuredT = normalizeMonitor(lsim.getMonitorAngle(transmittedIndex));
var hits = lsim.getMonitorHitsAll();
var measuredRp = hits[reflectedIndex]/PHOTONS;
var measuredTp = hits[transmittedIndex]/PHOTONS;
var missing = PHOTONS-hits[reflectedIndex]-hits[transmittedIndex];
var expectedRp = expectedProbability(lut, ANGLE, true);
var expectedTp = expectedProbability(lut, ANGLE, false);
var expectedR = lutMarginal(lut, ANGLE, true);
var expectedT = lutMarginal(lut, ANGLE, false);
var nTheta = lut.Binning.ThetaOutBins;
var nPhi = lut.Binning.PhiOutBins;
var measuredRJoint = monitorJoint(
    lsim.getMonitorAnglePhi(reflectedIndex), nTheta, nPhi, reverse);
var measuredTJoint = monitorJoint(
    lsim.getMonitorAnglePhi(transmittedIndex), nTheta, nPhi, reverse);
var expectedRJoint = lutJoint(lut, ANGLE, true);
var expectedTJoint = lutJoint(lut, ANGLE, false);
var rJointTvd = totalVariation(measuredRJoint, expectedRJoint);
var tJointTvd = totalVariation(measuredTJoint, expectedTJoint);
var rJointLimit = noiseTvdLimit(expectedRJoint, hits[reflectedIndex]);
var tJointLimit = noiseTvdLimit(expectedTJoint, hits[transmittedIndex]);
var rProbabilityLimit = probabilityLimit(expectedRp, PHOTONS);
var tProbabilityLimit = probabilityLimit(expectedTp, PHOTONS);
var passed = Math.abs(measuredRp-expectedRp) <= rProbabilityLimit &&
             Math.abs(measuredTp-expectedTp) <= tProbabilityLimit &&
             Math.abs(missing)/PHOTONS <= 0.001 &&
             rJointTvd <= rJointLimit && tJointTvd <= tJointLimit;

// Blue/green guides show representative phi=0 rays at the measured mean theta.
var meanR = meanAngle(measuredR)*Math.PI/180.0;
var meanT = meanAngle(measuredT)*Math.PI/180.0;
var reflectedDirection = add(
    scale(EX, Math.sin(meanR)),
    scale(NORMAL, -normalSign*Math.cos(meanR)));
var transmittedDirection = add(
    scale(EX, Math.sin(meanT)),
    scale(NORMAL, normalSign*Math.cos(meanT)));
geowin.addTrack([[0, 0, 0], scale(reflectedDirection, displayLength)], 4, 1, 3);
geowin.addTrack([[0, 0, 0], scale(transmittedDirection, displayLength)], 3, 1, 3);
geowin.showTracksAndMarkers();

var names = ["Rotated_LUT_reflected", "Rotated_monitor_reflected",
             "Rotated_LUT_transmitted", "Rotated_monitor_transmitted"];
for (var i = 0; i < names.length; i++) graph.remove(names[i]);
graph.new1D(names[0]); graph.addPoints(names[0], expectedR);
graph.new1D(names[1]); graph.addPoints(names[1], measuredR);
graph.new1D(names[2]); graph.addPoints(names[2], expectedT);
graph.new1D(names[3]); graph.addPoints(names[3], measuredT);
graph.setTitle(names[0], "LUT reflected");
graph.setTitle(names[1], "rotated monitor reflected");
graph.setTitle(names[2], "LUT transmitted");
graph.setTitle(names[3], "rotated monitor transmitted");
for (var i = 0; i < names.length; i++)
    graph.setAxisTitles(names[i], "theta_out from rotated interface normal, deg",
                       "conditional probability density, 1/deg");
graph.setLineProperties(names[0], 4, 1, 3);
graph.setMarkerProperties(names[1], 4, 20, 0.8);
graph.setLineProperties(names[2], 3, 1, 3);
graph.setMarkerProperties(names[3], 3, 21, 0.8);
graph.setMinimum(names[0], 0);
graph.draw(names[0], "AL");
graph.draw(names[1], "Psame");
graph.draw(names[2], "Lsame");
graph.draw(names[3], "Psame");
grwin.addLegend(0.55, 0.66, 0.92, 0.90,
                DIRECTION + ", " + ANGLE + " deg, rotated");
grwin.addToBasket("Rotated DavisLUT validation " + DIRECTION);
grwin.showWindow();

var jointTag = DIRECTION + "_" + String(ANGLE).replace(".", "p") + "_rotated";
var jointNames = ["Rotated_LUT_joint_reflected", "Rotated_monitor_joint_reflected",
                  "Rotated_LUT_joint_transmitted", "Rotated_monitor_joint_transmitted"];
for (var i = 0; i < jointNames.length; i++) hist.remove(jointNames[i]);
makeJointHistogram(jointNames[0], expectedRJoint, nTheta, nPhi,
                   "LUT expected reflected;rotated " + DIRECTION);
makeJointHistogram(jointNames[1], measuredRJoint, nTheta, nPhi,
                   "Monitor reflected;rotated " + DIRECTION);
makeJointHistogram(jointNames[2], expectedTJoint, nTheta, nPhi,
                   "LUT expected transmitted;rotated " + DIRECTION);
makeJointHistogram(jointNames[3], measuredTJoint, nTheta, nPhi,
                   "Monitor transmitted;rotated " + DIRECTION);
matchJointScales(jointNames[0], jointNames[1], expectedRJoint, measuredRJoint);
matchJointScales(jointNames[2], jointNames[3], expectedTJoint, measuredTJoint);
showAndSaveJoint(jointNames[0], "Rotated LUT reflected " + jointTag,
                 OUT + "/joint_" + jointTag + "_reflected_expected.png");
showAndSaveJoint(jointNames[1], "Rotated monitor reflected " + jointTag,
                 OUT + "/joint_" + jointTag + "_reflected_measured.png");
showAndSaveJoint(jointNames[2], "Rotated LUT transmitted " + jointTag,
                 OUT + "/joint_" + jointTag + "_transmitted_expected.png");
showAndSaveJoint(jointNames[3], "Rotated monitor transmitted " + jointTag,
                 OUT + "/joint_" + jointTag + "_transmitted_measured.png");
grwin.showWindow();

core.print("Rotated frame EX", EX, "EY", EY, "N", NORMAL);
core.print("Rotated GUI validation", DIRECTION, "angle", ANGLE,
           "R measured/expected", measuredRp, expectedRp,
           "T measured/expected", measuredTp, expectedTp,
           "missing", missing);
core.print("Probability limits R", rProbabilityLimit,
           Math.abs(measuredRp-expectedRp) <= rProbabilityLimit ? "PASS" : "FAIL",
           "T", tProbabilityLimit,
           Math.abs(measuredTp-expectedTp) <= tProbabilityLimit ? "PASS" : "FAIL");
core.print("Joint theta/phi TVD R", rJointTvd, "limit", rJointLimit,
           rJointTvd <= rJointLimit ? "PASS" : "FAIL",
           "T", tJointTvd, "limit", tJointLimit,
           tJointTvd <= tJointLimit ? "PASS" : "FAIL");
core.print("ROTATED GUI VALIDATION", passed ? "PASS" : "FAIL");
core.print("Geometry viewer: red=incident, blue=mean reflected, green=mean transmitted");
core.print("Reload validate_geom_gui.json to restore the unrotated setup");
core.saveObject({
    passed: passed,
    direction: DIRECTION,
    incidenceAngle: ANGLE,
    rotationAngles: ROTATION,
    ex: EX,
    ey: EY,
    normal: NORMAL,
    photons: PHOTONS,
    reflected: {
        measuredProbability: measuredRp,
        expectedProbability: expectedRp,
        probabilityLimit: rProbabilityLimit,
        jointTvd: rJointTvd,
        jointTvdLimit: rJointLimit
    },
    transmitted: {
        measuredProbability: measuredTp,
        expectedProbability: expectedTp,
        probabilityLimit: tProbabilityLimit,
        jointTvd: tJointTvd,
        jointTvdLimit: tJointLimit
    },
    missing: missing
}, OUT + "/metrics_" + jointTag + ".json");

function add(a, b)
{
    return [a[0]+b[0], a[1]+b[1], a[2]+b[2]];
}

function scale(a, value)
{
    return [value*a[0], value*a[1], value*a[2]];
}

function dot(a, b)
{
    return a[0]*b[0] + a[1]*b[1] + a[2]*b[2];
}

function cross(a, b)
{
    return [a[1]*b[2]-a[2]*b[1],
            a[2]*b[0]-a[0]*b[2],
            a[0]*b[1]-a[1]*b[0]];
}

function rotateAxis(vector, axis, angleDegrees)
{
    var angle = angleDegrees*Math.PI/180.0;
    var c = Math.cos(angle), s = Math.sin(angle);
    return add(add(scale(vector, c), scale(cross(axis, vector), s)),
               scale(axis, dot(axis, vector)*(1.0-c)));
}

// Matches AGeometryHub/TGeo: Phi about global z, Theta about the Phi-rotated
// x axis, then Psi about the resulting local z axis.
function rotateVector(vector, angles)
{
    var z = [0, 0, 1];
    var xAfterPhi = rotateAxis([1, 0, 0], z, angles[0]);
    var result = rotateAxis(vector, z, angles[0]);
    result = rotateAxis(result, xAfterPhi, angles[1]);
    var zAfterTheta = rotateAxis(z, xAfterPhi, angles[1]);
    return rotateAxis(result, zAfterTheta, angles[2]);
}

function normalizeMonitor(data)
{
    var total = 0;
    for (var i = 0; i < data.length; i++) total += data[i][1];
    var width = data.length > 1 ? data[1][0]-data[0][0] : 1.0;
    var result = [];
    for (var i = 0; i < data.length; i++)
        result.push([data[i][0], total > 0 ? data[i][1]/total/width : 0]);
    return result;
}

function meanAngle(distribution)
{
    var sum = 0, norm = 0;
    for (var i = 0; i < distribution.length; i++)
    {
        sum += distribution[i][0]*distribution[i][1];
        norm += distribution[i][1];
    }
    return norm > 0 ? sum/norm : 0;
}

function incidenceMix(data, angle)
{
    var n = data.Binning.ThetaIncBins;
    var x = angle/(90.0/n)-0.5;
    var k0 = Math.floor(x), f = x-k0;
    if (k0 < 0) { k0 = 0; f = 0; }
    if (k0 >= n-1) { k0 = n-1; f = 0; }
    return {k0:k0, k1:f > 0 ? k0+1 : k0, f:f};
}

function expectedProbability(data, angle, reflected)
{
    var mix = incidenceMix(data, angle);
    var counts = reflected ? data.ReflectedCounts : data.TransmittedCounts;
    return (1.0-mix.f)*counts[mix.k0]/data.Launched[mix.k0] +
           mix.f*counts[mix.k1]/data.Launched[mix.k1];
}

function lutMarginal(data, angle, reflected)
{
    var mix = incidenceMix(data, angle);
    var nt = data.Binning.ThetaOutBins, np = data.Binning.PhiOutBins;
    var source = reflected ? data.ReflectedHist : data.TransmittedHist;
    var probability = expectedProbability(data, angle, reflected);
    var width = 90.0/nt, result = [];
    for (var it = 0; it < nt; it++)
    {
        var value = 0;
        for (var ip = 0; ip < np; ip++)
        {
            var j = it*np+ip;
            value += (1.0-mix.f)*source[mix.k0][j]/data.Launched[mix.k0];
            if (mix.k1 != mix.k0)
                value += mix.f*source[mix.k1][j]/data.Launched[mix.k1];
        }
        result.push([(it+0.5)*width,
                     probability > 0 ? value/probability/width : 0]);
    }
    return result;
}

function lutJoint(data, angle, reflected)
{
    var mix = incidenceMix(data, angle);
    var size = data.Binning.ThetaOutBins*data.Binning.PhiOutBins;
    var source = reflected ? data.ReflectedHist : data.TransmittedHist;
    var probability = expectedProbability(data, angle, reflected);
    var result = [];
    for (var j = 0; j < size; j++)
    {
        var value = (1.0-mix.f)*source[mix.k0][j]/data.Launched[mix.k0];
        if (mix.k1 != mix.k0)
            value += mix.f*source[mix.k1][j]/data.Launched[mix.k1];
        result.push(probability > 0 ? value/probability : 0);
    }
    return result;
}

function monitorJoint(data, nTheta, nPhi, mirrorPhi)
{
    if (data.length != nTheta*nPhi)
        core.abort("Monitor joint binning does not match LUT: got " + data.length);
    var result = [], total = 0;
    for (var j = 0; j < nTheta*nPhi; j++) result.push(0);
    for (var it = 0; it < nTheta; it++)
        for (var ip = 0; ip < nPhi; ip++)
        {
            var targetPhi = mirrorPhi ? nPhi-1-ip : ip;
            var count = data[it*nPhi+ip][2];
            result[it*nPhi+targetPhi] += count;
            total += count;
        }
    if (total > 0)
        for (var j = 0; j < result.length; j++) result[j] /= total;
    return result;
}

function totalVariation(measured, expected)
{
    var sum = 0;
    for (var i = 0; i < expected.length; i++)
        sum += Math.abs(measured[i]-expected[i]);
    return 0.5*sum;
}

function noiseTvdLimit(expected, samples)
{
    if (samples < 1) return 1.0;
    var mean = 0;
    for (var i = 0; i < expected.length; i++)
        mean += Math.sqrt(Math.max(
            0, 2.0*expected[i]*(1.0-expected[i])/(Math.PI*samples)));
    return Math.max(0.01, 1.5*mean);
}

function probabilityLimit(expected, samples)
{
    return Math.max(
        0.001,
        6.0*Math.sqrt(Math.max(expected*(1.0-expected), 1.0/samples)/samples));
}

function makeJointHistogram(name, distribution, nTheta, nPhi, title)
{
    hist.new2D(name, nTheta, 0, 90, nPhi, 0, 360);
    for (var it = 0; it < nTheta; it++)
        for (var ip = 0; ip < nPhi; ip++)
            hist.fill(name, (it+0.5)*90.0/nTheta, (ip+0.5)*360.0/nPhi,
                      distribution[it*nPhi+ip]);
    hist.setTitle(name, title);
    hist.setAxisTitles(name, "theta_out in local frame, deg",
                       "phi_out relative to incidence plane, deg",
                       "conditional probability per bin");
    hist.setMinimum(name, 0);
}

function matchJointScales(expectedName, measuredName, expected, measured)
{
    var maximum = 0;
    for (var i = 0; i < expected.length; i++)
        maximum = Math.max(maximum, expected[i], measured[i]);
    hist.setMaximum(expectedName, maximum);
    hist.setMaximum(measuredName, maximum);
}

function showAndSaveJoint(name, basketTitle, fileName)
{
    hist.draw(name, "colz");
    grwin.addToBasket(basketTitle);
    grwin.saveImage(fileName);
}
